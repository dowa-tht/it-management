procedure Service_CheckSLAThreshold;
var
  Incs : array of TIncident;
  i    : integer;
begin
  Incs := Service_GetIncidents(Filter:(Status:Open));
  for i := 0 to Length(Incs)-1 do
    if (Incs[i].SLADeadline - Now) < (0.1 * 24*60*60) then   // 10% ของวันทำงานเหลืออยู่
      PushAlert('Incident '+Incs[i].CaseNumber+' ใกล้หมด SLA');
end;
