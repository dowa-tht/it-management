type
  // ---------- ผู้ใช้ ----------
  TUser = record
    Id       : string;
    FullName : string;
    Email    : string;
    Role     : (SuperUser, Admin, User);
  end;

  // ---------- การตั้งค่า ----------
  TWorkingHours = record
    StartTime : string;                     // '08:30'
    EndTime   : string;                     // '17:30'
    WorkDays  : set of (Mon,Tue,Wed,Thu,Fri,Sat,Sun);
  end;

  THoliday = record
    Date   : TDateTime;                     // วันหยุดเต็มวัน
    NameTH : string;                        // ชื่อภาษาไทย
    NameEN : string;                        // ชื่อภาษาอังกฤษ
  end;

  // ---------- การอธิบาย SLA ----------
  TExclusion = record
    Id        : string;
    Reason    : string;                     // เหตุผล pause SLA
    StartAt   : TDateTime;                  // เวลาเริ่ม pause
    EndAt     : TDateTime;                  // เวลาจบ (null = ยังเปิด)
  end;

  // ---------- Incident ----------
  TIncident = record
    Id                : string;
    CaseNumber        : string;
    Title             : string;
    Description       : string;
    Severity          : (High, Medium, Low);
    Status            : (Open, InProgress, Resolved, Closed);
    ReportedBy        : TUser;
    AssignedTo        : TUser;
    CreatedAt         : TDateTime;
    UpdatedAt         : TDateTime;
    ResolvedAt        : TDateTime;          // nullable
    SLAStartAt        : TDateTime;          // เวลาเริ่มนับ SLA
    SLADeadline       : TDateTime;          // คำนวณจาก SLAStartAt + NetBusinessMinutes
    SLAComplianceRate : real;               // % ของ SLA ที่ทำสำเร็จ
    Exclusions        : array of TExclusion; // รายการ manual pause
  end;

  // ---------- KPI Card ----------
  TStatCard = record
    Label : string;
    Value : integer;
    Color : string;    // สี Hex
    Link  : string;    // URL ไปหน้าที่เกี่ยวข้อง
  end;
