type
  // -------------------  Core Entities -------------------
  TUser = record
    id          : string;
    fullName    : string;
    email       : string;
    role        : (SuperUser, Admin, User);
  end;

  TIncident = record
    id                 : string;
    caseNumber         : string;
    title              : string;
    description        : string;
    severity           : (High, Medium, Low);
    status             : (Open, InProgress, Resolved, Closed);
    reportedBy         : TUser;
    assignedTo         : TUser;
    createdAt          : TDateTime;
    updatedAt          : TDateTime;
    resolvedAt         : TDateTime;   // nullable
    slaStartAt         : TDateTime;   // เวลาเริ่ม SLA (หลังจากสร้าง)
    slaDeadline        : TDateTime;   // คำนวณจาก slaStartAt + NetBusinessMinutes
    slaComplianceRate : real;        // % ของ SLA ที่สำเร็จ
    exclusions         : array of TExclusion; // manual pause บน incident
  end;

  // -------------------  Settings -----------------------
  TWorkingHours = record
    startTime  : string;   // HH:MM (24‑h)  เช่น '08:30'
    endTime    : string;   // HH:MM          เช่น '17:30'
    workDays   : set of (Mon, Tue, Wed, Thu, Fri, Sat, Sun);
  end;

  THoliday = record
    date      : TDateTime;            // วันหยุด (เต็มวัน)
    nameTH    : string;               // ชื่อวันหยุดภาษาไทย
    nameEN    : string;               // ชื่อวันหยุดภาษาอังกฤษ
  end;

  TExclusion = record
    id        : string;
    reason    : string;               // เหตุผลที่ pause SLA
    startAt   : TDateTime;            // เวลาเริ่ม pause
    endAt     : TDateTime;            // เวลาจบ pause (nullable – ถ้ายังเปิดอยู่)
  end;

  // -------------------  KPI / Statistics ---------------
  TStatCard = record
    label   : string;       // ตัวอย่าง: 'Open Incidents'
    value   : integer;      // จำนวน
    color   : string;       // สีของ card
    link    : string;       // URL ไปยังหน้าที่เกี่ยวข้อง
  end;
