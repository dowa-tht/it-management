procedure OpenIncidentDetail(IncId: string);
var
  Inc : TIncident;
begin
  Inc := Service_GetIncidentById(IncId);
  Render('IncidentDetail', Inc);
end;

// ปรับสถานะ/มอบหมาย
procedure UI_UpdateIncident(Inc: TIncident);
begin
  if Service_SaveIncident(Inc) then
    ShowToast('บันทึก Incident สำเร็จ')
  else
    ShowError('บันทึก Incident ไม่สำเร็จ');
end;

// SLA Pause / Resume / Manual Exclusion
procedure UI_AddExclusion(IncId: string; Excl: TExclusion);
begin
  if Service_AddExclusion(IncId, Excl) then
    ShowToast('เพิ่มการพัก SLA สำเร็จ')
  else
    ShowError('เพิ่มการพัก SLA รายการล้มเหลว');
end;

procedure UI_ResumeSLA(IncId: string);
begin
  if Service_ResumeSLA(IncId) then
    ShowToast('Resume SLA สำเร็จ')
  else
    ShowError('Resume SLA ไม่สำเร็จ');
end;
