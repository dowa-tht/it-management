function Service_AddHoliday(H: THoliday): boolean;
begin
  Result := DB_Insert('holidays', JSON_Serialize(H));
end;

function Service_UpdateHoliday(Id: string; H: THoliday): boolean;
begin
  Result := DB_Upsert('holidays', Id, JSON_Serialize(H));
end;

function Service_DeleteHoliday(Id: string): boolean;
begin
  Result := DB_Delete('holidays', Id);
end;

function Service_GetAllHolidays: array of THoliday;
var
  rs : variant;
  i  : integer;
begin
  rs := DB_Query('SELECT * FROM holidays ORDER BY date');
  SetLength(Result, rs.RowCount);
  for i := 0 to rs.RowCount-1 do
    Result[i] := JSON_Deserialize<THoliday>(rs[i].json);
end;
